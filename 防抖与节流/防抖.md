            `console.log('123',this,temp);`

        `function debounce(fn,delay){`
            `let timer = null;`
            `return function (...arg) {`
                `if (timer) {`
                    `clearTimeout(timer);`
                `}`
                `timer = setTimeout(()=>{`
                    `fn.apply(this, arg);`
                    `timer = null;`
                `},delay)`
            `}`
        `}`
`const debounceClick123 = debounce(click123, 3000);`
`document.getElementById('123').addEventListener('click', debounceClick123);`
    `</script>`
`</body>`

每次触发事件时，先清除之前定时器，再重新设置新的定时器；只有当停止触发、且等待时间达到设定的 `delay` 后，最后一次设置的定时器才会执行回调，实现「仅最后一次触发生效」。

防抖的核心是在事件触发后等待指定时间，如果期间再次触发就重新计时，只执行最后一次
