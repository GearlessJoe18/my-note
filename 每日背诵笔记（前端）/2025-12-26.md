### 对象中函数简写
```javascript
// 【传统写法】ES5 及之前
const person = {
  name: '张三',
  add: function(a, b) {
    return a + b;
  }
};

// 【ES6 简写写法】（推荐）
const person = {
  name: '张三',
  add(a, b) {
    return a + b;
  }
};

// 调用方式完全一样
console.log(person.add(1, 2)); // 输出：3
```
