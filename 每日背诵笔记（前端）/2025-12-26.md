### 对象中函数简写
```javascript
// 【传统写法】ES5 及之前
const person = {
  name: '张三',
  sayName: function() {
    console.log('我的名字是：' + this.name);
  },
  // 带参数的传统写法
  add: function(a, b) {
    return a + b;
  }
};

// 【ES6 简写写法】（推荐）
const person = {
  name: '张三',
  // 简写：直接 方法名() {}，省略 : function
  sayName() {
    console.log('我的名字是：' + this.name);
  },
  // 带参数的简写
  add(a, b) {
    return a + b;
  }
};

// 调用方式完全一样
console.log(person.sayName()); // 输出：我的名字是：张三
console.log(person.add(1, 2)); // 输出：3
```
