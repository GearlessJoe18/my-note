`function resolve(data) {`
        `if (this.MyPromiseState === 'pending') {`
            `self.MyPromiseState = 'fulfilled'`
            `self.MyPromiseResult = data`
        `}`
    `}`
`function reject(data) {`
	`if (self.MyPromiseState !== 'pending') {return}`
	`self.MyPromiseState = 'rejected'`
	`self.MyPromiseResult = data`
`}`
